cmake_minimum_required(VERSION 3.19)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(SDK "$ENV{PLAYDATE_SDK_PATH}")
if(NOT EXISTS "${SDK}/C_API/pd_api.h")
    message(FATAL_ERROR "Set PLAYDATE_SDK_PATH to the installed Playdate SDK")
endif()
set(PLAYDATE_GAME_NAME sdk_check)
set(PLAYDATE_GAME_DEVICE sdk_check_device)
project(sopwith_sdk_check C ASM)
if(TOOLCHAIN STREQUAL "armgcc")
    add_executable(${PLAYDATE_GAME_DEVICE} main.c)
else()
    add_library(${PLAYDATE_GAME_NAME} SHARED main.c)
endif()
include("${SDK}/C_API/buildsupport/playdate_game.cmake")
