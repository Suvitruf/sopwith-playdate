cmake_minimum_required(VERSION 3.19)
project(sopwith_playdate C ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(SOPWITH_TESTS "Build host tests instead of a Playdate bundle" OFF)
option(SOPWITH_SANITIZERS "Instrument host tests" OFF)
set(CORE_FILES swauto swcollsn swend swgames swgrpha swinit swmain swmove
    swobject swsound swsplat swstbar swsymbol swtext video)
foreach(name IN LISTS CORE_FILES)
    list(APPEND GAME_SOURCES Sources/core/${name}.c)
endforeach()
list(APPEND GAME_SOURCES Sources/port/engine.c Sources/port/runtime.c Sources/port/markers.c Sources/port/hud.c Sources/port/pause_panel.c
    Sources/port/profile.c Sources/port/tone.c)
include_directories(Sources/port Sources/core)
add_compile_definitions(SOPWITH_PORT=1)

if(SOPWITH_TESTS)
    enable_testing()
    add_executable(port_tests tests/test_port.c tests/test_hud.c tests/test_pause_panel.c ${GAME_SOURCES})
    target_compile_options(port_tests PRIVATE -Wall -Wextra -Wno-sign-compare -UNDEBUG)
    if(SOPWITH_SANITIZERS)
        target_compile_options(port_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(port_tests PRIVATE -fsanitize=address,undefined)
    endif()
    add_test(NAME port_tests COMMAND port_tests)
    add_test(NAME port_stress COMMAND port_tests --stress)
    set_tests_properties(port_stress PROPERTIES TIMEOUT 60)
    add_executable(profile_tests tests/test_profile.c Sources/port/profile.c Sources/port/tone.c Sources/port/markers.c)
    target_compile_options(profile_tests PRIVATE -Wall -Wextra -UNDEBUG)
    if(SOPWITH_SANITIZERS)
        target_compile_options(profile_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(profile_tests PRIVATE -fsanitize=address,undefined)
    endif()
    add_test(NAME profile_tests COMMAND profile_tests)
    set(TEST_SDK "$ENV{PLAYDATE_SDK_PATH}")
    if(NOT EXISTS "${TEST_SDK}/C_API/pd_api.h")
        set(TEST_SDK "${CMAKE_SOURCE_DIR}/.tools/PlaydateSDK-2.2.0")
    endif()
    if(EXISTS "${TEST_SDK}/C_API/pd_api.h")
        add_executable(app_tests tests/test_app.c ${GAME_SOURCES}
            Sources/playdate/main.c Sources/playdate/storage.c)
        target_include_directories(app_tests PRIVATE "${TEST_SDK}/C_API")
        target_compile_definitions(app_tests PRIVATE TARGET_EXTENSION=1 TARGET_SIMULATOR=1)
        target_compile_options(app_tests PRIVATE -Wall -Wextra -Wno-sign-compare -UNDEBUG)
        if(SOPWITH_SANITIZERS)
            target_compile_options(app_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(app_tests PRIVATE -fsanitize=address,undefined)
        endif()
        add_test(NAME app_tests COMMAND app_tests)
    else()
        message(STATUS "app_tests skipped: install SDK 2.2.0 or set PLAYDATE_SDK_PATH for API headers")
    endif()
    return()
endif()

set(SDK "$ENV{PLAYDATE_SDK_PATH}")
if(NOT EXISTS "${SDK}/C_API/pd_api.h")
    message(FATAL_ERROR "Set PLAYDATE_SDK_PATH, or use bash tools/build-game.sh")
endif()
set(PLAYDATE_GAME_NAME sopwith)
set(PLAYDATE_GAME_DEVICE sopwith_device)
if(TOOLCHAIN STREQUAL "armgcc")
    set(GAME_TARGET ${PLAYDATE_GAME_DEVICE})
    add_executable(${GAME_TARGET} ${GAME_SOURCES} Sources/playdate/main.c Sources/playdate/storage.c)
else()
    set(GAME_TARGET ${PLAYDATE_GAME_NAME})
    add_library(${GAME_TARGET} SHARED ${GAME_SOURCES} Sources/playdate/main.c Sources/playdate/storage.c)
    if(UNIX AND NOT APPLE)
        target_link_options(${GAME_TARGET} PRIVATE -Wl,--no-undefined)
    endif()
endif()
# Use external SDK compiler/startup settings, with packaging confined to this build tree.
include("${SDK}/C_API/buildsupport/playdate.cmake")
target_compile_options(${GAME_TARGET} PRIVATE -UNDEBUG)
set(CMAKE_VERBOSE_MAKEFILE OFF)
set(STAGING "${CMAKE_CURRENT_BINARY_DIR}/Source")
file(MAKE_DIRECTORY "${STAGING}")
configure_file(Source/pdxinfo "${STAGING}/pdxinfo" COPYONLY)
# Stage launcher art on every package pass, including incremental builds.
# Removing only this generated subdirectory avoids retaining deleted frames.
file(GLOB_RECURSE LAUNCHER_ASSETS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Source/launcher/*")
if(TOOLCHAIN STREQUAL "armgcc")
    set_target_properties(${GAME_TARGET} PROPERTIES SUFFIX ".elf")
    add_custom_command(TARGET ${GAME_TARGET} POST_BUILD
        COMMAND ${CMAKE_STRIP} --strip-unneeded -R .comment -g
            "$<TARGET_FILE:${GAME_TARGET}>" -o "${STAGING}/pdex.elf")
else()
    add_custom_command(TARGET ${GAME_TARGET} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_FILE:${GAME_TARGET}>"
            "${STAGING}/pdex${CMAKE_SHARED_LIBRARY_SUFFIX}")
endif()
add_custom_target(bundle ALL
    COMMAND ${CMAKE_COMMAND} -E rm -rf "${STAGING}/launcher"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/Source/launcher" "${STAGING}/launcher"
    COMMAND ${PDC} "${STAGING}" "${CMAKE_CURRENT_BINARY_DIR}/sopwith.pdx"
    DEPENDS ${GAME_TARGET} "${STAGING}/pdxinfo" ${LAUNCHER_ASSETS})
